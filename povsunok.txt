    <h3>Мій рівень спокою на початку року:</h3>
    
    <div class="slider-container">
        <input type="range" min="1" max="10" value="5" class="slider" id="calmSlider">
        <div class="value-display" id="sliderValue">5/10 😐</div>
    </div>

    <button class="btn" id="sendBtn">Зберегти результат</button>

    <script>
        // Ініціалізуємо Telegram WebApp
        const tg = window.Telegram.WebApp;
        tg.expand(); // Розгортаємо шторку на оптимальну висоту

        const slider = document.getElementById('calmSlider');
        const output = document.getElementById('sliderValue');
        const sendBtn = document.getElementById('sendBtn');

        // Емодзі під кожен рівень стресу/спокою
        const emojis = ["😭", "😰", "😰", "😬", "😐", "😐", "🙂", "😌", "😎", "🧘‍♂️"];

        // Оновлюємо цифру та емодзі при перетягуванні язичка
        slider.oninput = function() {
            const val = this.value;
            output.innerHTML = `${val}/10 ${emojis[val - 1]}`;
        };

        // Дія при натисканні на кнопку "Зберегти"
        sendBtn.onclick = function() {
            const resultData = {
                score: slider.value
            };
            
            // Відправляємо дані назад нашому боту в Telegram
            tg.sendData(JSON.stringify(resultData));
            
            // Закриваємо шторку додатка
            tg.close();
        };
    </script>
